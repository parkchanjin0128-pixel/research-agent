\section{발표효과}

\begin{frame}{한 푼도 쓰지 않은 정책이 시장을 되돌렸다}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.53\textwidth}
\lab{2020년 3월, 미국 회사채 시장}
\begin{itemize}
\item 3월 23일 연준이 회사채 매입기구 두 개(PMCCF, SMCCF)를 \textbf{발표}
\item 발표 시점에 투자등급 회사채 가격이 약 7\% 상승, 다른 자산은 거의 무반응
\item 실제 매입은 한참 뒤: SMCCF 5월 12일, PMCCF 6월 29일 개시
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\lab{실제로 쓴 돈}\par\vspace{1.5mm}
\begingroup\tf
\begin{tabularx}{\linewidth}{@{}Yr@{}}
\toprule
기구 & 최대 이용액\\
\midrule
SMCCF (유통시장) & 141억 달러\\
PMCCF (발행시장) & \textbf{0달러}\\
\bottomrule
\end{tabularx}\endgroup
\vspace{2.5mm}
{\small\RaggedRight PMCCF는 한 건도 매입하지 않고 종료됐다. 시장을 움직인 것은 매입이 아니라 \textbf{매입하겠다는 말}이었다.\par}
\end{column}
\end{columns}
\takeaway{발표만으로 가격이 움직이려면 조건이 하나 있다. 시장이 그 말을 \textbf{믿어야} 한다.}
\src{Haddad, Moreira \& Muir (2021, \textit{Review of Financial Studies}); Yale Program on Financial Stability, ``PMCCF and SMCCF''.}
\end{frame}

\begin{frame}{발표효과는 금융경제학의 핵심 주제다}
\begingroup\tfs
\begin{tabularx}{\textwidth}{@{}L{0.2\textwidth}L{0.34\textwidth}Y@{}}
\toprule
영역 & 발표 & 연구가 보인 것\\
\midrule
양적완화 & 2008--2010년 연준 자산매입 발표 & 매입 전, 발표일에 장기금리 하락 (Krishnamurthy \& Vissing-Jorgensen 2011)\\
포워드 가이던스 & FOMC 성명의 문구 & 금리 결정과 별개로 '미래 경로에 대한 말'이 가격을 움직임 (Gürkaynak et al. 2005)\\
유로 위기 & 2012년 ``whatever it takes''와 OMT & 발동된 적 없는 OMT가 남유럽 국채금리를 낮춤 (Altavilla et al. 2016)\\
코로나 위기 & 2020년 회사채 매입기구 & 발표 시점에 스프레드 축소, PMCCF 매입 0 (Haddad et al. 2021)\\
무역정책 & 2018--2019년 관세 발표 & 발표일 주가 하락과 이후 실물 성과 (Amiti et al.)\\
\bottomrule
\end{tabularx}\endgroup
\takeaway{공통점: 발표자가 \textbf{약속을 지킨다고 여겨지는 기관}이었다. 발표자가 반복해서 실행하지 않는 경우는 거의 다뤄지지 않았다.}
\src{서지 사항은 논문의 참고문헌 참조.}
\end{frame}

\src{본 연구의 사건 목록(가격 열람 전 고정). 실행률은 그 시점까지 결과가 확정된 과거 위협만으로 계산한 Beta(1,1) 사후평균. 목록은 전수가 아니다.}
\end{frame}

\begin{frame}{연구 질문}
\begin{itemize}
\item \textbf{질문}\; 실행되지 않는 위협이 반복될 때, 위협의 발표효과는 어떻게 되는가
\item \textbf{왜 중요한가}\; 위협은 실행 비용 없이 협상력을 얻는 수단이다. 시장이 믿지 않으면 그 수단은 사라진다
\item \textbf{왜 지금 가능한가}\; 2025년 이후 같은 발표자가 수십 번 위협했고, 각각의 실행 여부가 관측된다
\end{itemize}
\vspace{1mm}
\lab{세 가지 예측 (단순 학습 모형: 반응 $\approx \Delta p \times L$)}\par\vspace{1mm}
\begingroup\tf
\begin{tabularx}{\textwidth}{@{}lL{0.38\textwidth}Y@{}}
\toprule
 & 시장이 위협을 덜 믿으면 & 구분해 주는 것\\
\midrule
P1 & 위협 발표 시점의 반응이 작다 & 이것만으로는 '관세 뉴스에 무뎌진 것'과 구분되지 않음\\
P2 & 예정 발효 시점의 반응이 \textbf{크다} & 발표를 안 믿었으면 발효가 새 정보. 무뎌진 것이라면 여기도 작아야 함\\
P3 & 철회·완화 시점의 반응이 작다 & 처음에 덜 반영됐으면 되돌릴 것도 적음\\
\bottomrule
\end{tabularx}\endgroup
\takeaway{이 질문에 오기까지 두 번 실패했다. 그 과정이 지금의 방법을 정했다.}
\end{frame}

\section{첫 시도}

\begin{frame}{첫 시도: 2018년 수출통제 예고가 중국의 연구배분을 바꿨는가}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.5\textwidth}
\lab{원래 질문}
\begin{itemize}
\item 2018년 11월 미국이 신흥기술 수출통제를 \textbf{예고}(ANPRM)
\item 실제 통제 전에 중국이 해당 기술의 자체 연구를 늘렸는가
\end{itemize}
\vspace{1mm}
\lab{설계}
\begin{itemize}
\item 18기술 $\times$ 5국 $\times$ 120개월 특허 패널(10,800셀)
\item 지목 기술 대 비지목 기술, 사전 미국 의존도가 높을수록 반응이 큰지
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\lab{모형이 말해 준 것}\par\vspace{1mm}
{\small 규제 확률 $p$가 오르면 시장가치는 항상 하락하지만, 자체 연구 $I^*$의 방향은 정해지지 않는다.\par}
\vspace{-1.5mm}
\[\frac{dI^{*}}{dp}=-\frac{D\,R_{Ia}}{C''-R_{II}},\qquad \frac{dV}{dp}<0\]
\vspace{-2.5mm}
{\small\RaggedRight 미국 기술과 자체 연구가 대체재면 증가, 보완재면 감소. \textbf{주가만 보고는 연구의 방향을 알 수 없다.}\par}
\end{column}
\end{columns}
\takeaway{그래서 특허 자료로 연구배분을 직접 봐야 했다. 문제는 그 비교가 성립하지 않았다는 것이다.}
\end{frame}

\begin{frame}{실패 1: 비교할 기술이 없었다}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.54\textwidth}
\includegraphics[width=\linewidth]{fig/dose_ranges.pdf}
\end{column}
\begin{column}{0.42\textwidth}
\begin{itemize}
\item 지목 기술: AI·음성인식·적층제조 등
\item 비지목 기술: 성형·절삭·연삭
\item 미국 의존도 범위가 \textbf{한 곳도 겹치지 않는다}
\item 대체 지표로 겹치는 구간을 만들면 지목 2, 비지목 3단위만 남는다
\end{itemize}
\end{column}
\end{columns}
\takeaway{``같은 의존도의 지목 기술과 비지목 기술''이라는 비교 자체가 자료에 없었다. 지목된 기술은 애초에 다른 종류의 기술이었다.}
\src{$B^{\mathrm{cit}}$: 2013--2017년 중국 주소 출원인의 WO 문헌이 인용한 과거 문헌 중 미국 비중. 알려진 국가 인용 20쌍 이상인 기술.}
\end{frame}

\begin{frame}{실패 2: 아무 일도 없던 해에 ``효과''가 더 컸다}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.52\textwidth}
\includegraphics[width=\linewidth]{fig/fake_windows.pdf}
\end{column}
\begin{column}{0.44\textwidth}
\begin{itemize}
\item 실제 창(2018.12--2019.4)의 추정치: $-0.359$
\item 신호가 없던 2013--2015년 같은 달 창: $-1.1$에서 $-1.3$
\item 의존도를 4,999번 재배정해도 실제 추정치는 이례적이지 않았다
\end{itemize}
\end{column}
\end{columns}
\takeaway{추정치가 잡은 것은 정책이 아니라 중국의 \textbf{추격 추세}였다. 의존도가 높은 기술일수록 원래 빠르게 따라잡고 있었다.}
\src{단위: 사전 의존도 10\%p당 상대 $\log(1+\text{WO family})$ 기울기의 지목$-$비지목 차이, 2017년 기준. 공통 점범위 표본.}
\end{frame}

\begin{frame}{실패 3: 신호 이전부터 경로가 흔들렸다}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.56\textwidth}
\includegraphics[width=\linewidth]{fig/fig4_theta_orig.png}
\end{column}
\begin{column}{0.4\textwidth}
\begin{itemize}
\item 2010--2017년 사이 차등 기울기가 $+3$에서 $-2$까지 움직인다
\item 2018년 신호(세로선) 전후로 달라진 패턴이 보이지 않는다
\item 5개월 창으로 특허 출원의 반응을 잡기에는 연구에서 출원까지의 시차도 길다
\end{itemize}
\end{column}
\end{columns}
\takeaway{판정: 이 설계로는 연구배분의 인과효과를 말할 수 없다. 효과가 없다는 증거도 아니다.}
\src{월별 지목$-$비지목 차등 기울기. 파란 선 비절단 진단, 주황 선 공통 점범위 진단. 음영은 후보 창. (원 분석의 그림을 그대로 사용)}
\end{frame}

\begin{frame}{첫 시도에서 얻은 것}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\lab{버린 것}
\begin{itemize}
\item 특허로 연구배분을 재는 설계
\item 특허 인용 비중을 '미국 의존도'로 쓰는 지표
\item ``기술 대응''이라는 결과 변수
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\lab{남긴 것}
\begin{itemize}
\item 신호는 한 번에 오지 않는다: 보고서, 입법, 예고, 시행이 따로 도착한다
\item \textbf{예고와 시행을 구분}해야 한다
\item 가짜 시점·재배정으로 설계를 먼저 깨 보는 절차
\end{itemize}
\end{column}
\end{columns}
\takeaway{질문을 줄였다. ``예고가 연구를 바꿨는가'' 대신, 그 앞 단계인 ``예고가 정보로서 시장에 도착했는가''부터 보기로 했다.}
\end{frame}

\item 10건은 번복 발표도 없이 사라졌다. 시장은 이 위협들을 발표 때부터 거의 무시했다
\item 가장 크게 반응한 위협이 실행되지 않았다
\end{itemize}
\end{column}
\end{columns}
\takeaway{시장은 실행될 위협을 가려내지는 못했지만, 판이 큰 위협은 가려냈다. 실행률 22\%는 신뢰도의 척도가 아니다.}
\end{frame}

\begin{frame}{결론}
\begin{itemize}
\item \textbf{말뿐인 위협도 시장을 움직인다.} 20건 중 16건이 위험회피 방향이고, 철회에는 반대 방향으로 반응한다
\item \textbf{시장은 믿거나 안 믿는 것이 아니라 가려 듣는다.} 반응은 실행 여부가 아니라 판의 크기를 따랐고, 조용히 사라질 위협은 발표 때부터 무시됐다
\item \textbf{실행률은 신뢰도가 아니다.} 미실행에는 통한 위협과 무시된 위협이 섞여 있고, 예측시장은 실행 확률을 0.4 안팎으로 맞게 매겼다
\item \textbf{발효일은 사건이 아니다.} 정보는 말이 나오는 순간과 거둬지는 순간에 도착한다
\end{itemize}
\lab{한계}
\begin{itemize}
\item 시각을 확인한 사건이 강화 20, 완화 9, 발효 11건뿐이다. 집단 비교는 유의하지 않고 결말별 분류는 사후적이며, 큰 반응과 후퇴의 인과 방향은 가르지 못한다
\end{itemize}
\lab{남은 질문}
\begin{itemize}
\item 예고만으로 실제 연구개발 지출이 옮겨 갔는가. 주가가 보여 준 것은 기대까지다
\end{itemize}
\takeaway{PMCCF는 높은 신뢰로, 관세 위협은 낮은 실행률로도 말이 가격을 움직였다. 공통점은 실행 실적이 아니라 실행 능력과 판의 크기다.}
\end{frame}
